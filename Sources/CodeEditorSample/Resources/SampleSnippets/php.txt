<?php
// Greets the named person and returns the rendered string.
function greet(string $name, int $times = 1): string {
    $lines = [];
    for ($i = 0; $i < $times; $i++) {
        $lines[] = "Hello, {$name}! (" . ($i + 1) . ")";
    }
    return implode("\n", $lines);
}

$names = ["Ada", "Grace", "Linus"];
foreach ($names as $name) {
    echo greet($name, 2) . "\n";
}
