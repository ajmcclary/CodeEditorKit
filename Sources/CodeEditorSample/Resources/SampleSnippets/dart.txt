// Sample Dart program for syntax highlighting.
String greet(String name, {int times = 1}) {
    final lines = <String>[];
    for (var i = 0; i < times; i++) {
        lines.add("Hello, $name! (${i + 1})");
    }
    return lines.join("\n");
}

void main() {
    final names = ["Ada", "Grace", "Linus"];
    for (final name in names) {
        print(greet(name, times: 2));
    }
}
