/// Greets the named person and returns the rendered string.
fn greet(name: &str, times: u32) -> String {
    let mut out = String::new();
    for i in 0..times {
        out.push_str(&format!("Hello, {}! ({})\n", name, i + 1));
    }
    out
}

fn main() {
    let names = ["Ada", "Grace", "Linus"];
    for name in names.iter() {
        print!("{}", greet(name, 2));
    }
}
