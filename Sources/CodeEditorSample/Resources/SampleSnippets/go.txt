package main

import "fmt"

// Greet returns the rendered greeting string.
func Greet(name string, times int) string {
    out := ""
    for i := 0; i < times; i++ {
        out += fmt.Sprintf("Hello, %s! (%d)\n", name, i+1)
    }
    return out
}

func main() {
    names := []string{"Ada", "Grace", "Linus"}
    for _, name := range names {
        fmt.Print(Greet(name, 2))
    }
}
